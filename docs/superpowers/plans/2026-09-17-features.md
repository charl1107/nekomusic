static List<Song> getMostPlayed({int limit = 10}) {
  final entries = _playCountsBox.toMap().entries.toList()
    ..sort((a, b) => (b.value as int).compareTo(a.value as int));
  final recentData = _recentsBox.get('recents', defaultValue: '[]');
  final allSongs = (json.decode(recentData as String) as List)
      .map((s) => Song.fromJson(s))
      .toList();
  final songMap = {for (var s in allSongs) s.id: s};
  final result = <Song>[];
  for (var entry in entries.take(limit)) {
    if (songMap.containsKey(entry.key)) {
      result.add(songMap[entry.key]!);
    }
  }
  return result;
}

static List<Song> getDownloads() {
  final data = _downloadsBox.get('downloads', defaultValue: '[]');
  final list = json.decode(data as String) as List;
  return list.map((s) => Song.fromJson(s)).toList();
}

static Future<void> addDownload(Song song) async {
  final downloads = getDownloads();
  if (downloads.any((s) => s.id == song.id)) return;
  downloads.insert(0, song);
  await _downloadsBox.put('downloads',
      json.encode(downloads.map((s) => s.toJson()).toList()));
}

static Future<void> removeDownload(String songId) async {
  final downloads = getDownloads();
  downloads.removeWhere((s) => s.id == songId);
  await _downloadsBox.put('downloads',
      json.encode(downloads.map((s) => s.toJson()).toList()));
}

static bool isDownloaded(String songId) =>
    getDownloads().any((s) => s.id == songId);

import 'dart:async';
import 'package:flutter/foundation.dart';
import '../models/song.dart';
import '../services/api_service.dart';
import '../services/storage_service.dart';

enum DownloadState { idle, downloading, downloaded }

class DownloadService extends ChangeNotifier {
  final Map<String, DownloadState> _states = {};

  DownloadState getState(String songId) =>
      _states[songId] ?? (StorageService.isDownloaded(songId)
          ? DownloadState.downloaded
          : DownloadState.idle);

  Future<void> downloadSong(Song song) async {
    if (getState(song.id) == DownloadState.downloading) return;

    _states[song.id] = DownloadState.downloading;
    notifyListeners();

    try {
      final audioUrl = await ApiService.getAudioUrl(song.id);
      if (audioUrl == null) {
        _states[song.id] = DownloadState.idle;
        notifyListeners();
        return;
      }

      if (kIsWeb) {
        await _downloadWeb(audioUrl, song);
      } else {
        await _downloadMobile(audioUrl, song);
      }

      await StorageService.addDownload(song);
      _states[song.id] = DownloadState.downloaded;
    } catch (e) {
      _states[song.id] = DownloadState.idle;
    }
    notifyListeners();
  }

  Future<void> _downloadWeb(String url, Song song) async {
    // Uses dart:html - implemented in Task 3b
  }

  Future<void> _downloadMobile(String url, Song song) async {
    // Uses dio + path_provider - implemented in Task 3c
  }

  Future<void> removeDownload(String songId) async {
    await StorageService.removeDownload(songId);
    _states[songId] = DownloadState.idle;
    notifyListeners();
  }
}

def get_playlist(self, playlist_id):
    if not self.api_key:
        return self._mock_playlist(playlist_id)
    try:
        results = []
        next_page_token = None
        while True:
            params = {
                "part": "snippet",
                "playlistId": playlist_id,
                "maxResults": 50,
                "key": self.api_key,
            }
            if next_page_token:
                params["pageToken"] = next_page_token
            response = requests.get(
                f"{self.base_url}/playlistItems",
                params=params,
                timeout=10,
            )
            response.raise_for_status()
            data = response.json()
            for item in data.get("items", []):
                snippet = item["snippet"]
                video_id = snippet.get("resourceId", {}).get("videoId")
                if video_id:
                    results.append({
                        "id": video_id,
                        "title": snippet["title"],
                        "artist": snippet["channelTitle"],
                        "thumbnail": snippet["thumbnails"]["high"]["url"],
                        "duration": None,
                    })
            next_page_token = data.get("nextPageToken")
            if not next_page_token:
                break
        if results:
            video_ids = [r["id"] for r in results]
            durations = self._get_durations(video_ids)
            for r in results:
                r["duration"] = durations.get(r["id"])
        return {"results": results}
    except requests.RequestException:
        return self._mock_playlist(playlist_id)

static Future<List<Song>> importYouTubePlaylist(String url) async {
  final playlistId = _extractPlaylistId(url);
  if (playlistId == null) return [];

  try {
    final response = await http.get(
      Uri.parse('$baseUrl/playlist/$playlistId'),
    );
    if (response.statusCode == 200) {
      final data = json.decode(response.body);
      return (data['results'] as List)
          .map((json) => Song.fromJson(json))
          .toList();
    }
  } catch (e) {
    // ignore
  }
  return [];
}

static String? _extractPlaylistId(String url) {
  final uri = Uri.tryParse(url);
  if (uri == null) return null;
  return uri.queryParameters['list'];
}
